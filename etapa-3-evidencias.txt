import { PlayCircle } from 'lucide-react';
import { PageHeader } from '../components/DesignSystem';
import { ConnectYoutubeStep } from '../components/english/daily/ConnectYoutubeStep';
import { PlaylistResolutionPanel } from '../components/english/daily/PlaylistResolutionPanel';
import { RecentVideosPanel } from '../components/english/daily/RecentVideosPanel';
import { useAuth } from '../contexts/AuthContext';
import { useDailyVideoConnection } from '../hooks/useDailyVideoConnection';

// Página da Etapa 3 da V1 "Inglês Diário" — registrada em /estudo/ingles-diario
// (rota nova e independente, ver src/App.tsx e src/layouts/Layout.tsx). Não
// substitui nem altera src/pages/Ingles.tsx (Modo Entrevista), que continua
// intocado em /estudo/ingles.
export function InglesDiarioPage() {
  const { user } = useAuth();
  const connection = useDailyVideoConnection(user?.id ?? null);

  const isConnected = connection.authState.status === 'authorized';

  // Só mostra a lista de vídeos depois que uma playlist está confirmada
  // (validação bem-sucedida ou descoberta com exatamente uma encontrada).
  const playlistConfirmed =
    connection.validation?.kind === 'valid' || connection.discovery?.kind === 'found';

  const selectedVideo = connection.videos?.kind === 'videos'
    ? connection.videos.videos.find(video => video.videoId === connection.selectedVideoId) ?? null
    : null;

  return (
    <div className="space-y-6">
      <PageHeader
        eyebrow="Inglês Diário"
        title="Vídeo de hoje"
        subtitle="Conecte sua conta do YouTube, localize a playlist 'SGP — Inglês' e escolha o vídeo que você estudou hoje."
      />

      {!isConnected && (
        <ConnectYoutubeStep
          authState={connection.authState}
          onConnect={connection.connect}
          onConnectWithConsent={connection.connectWithConsent}
        />
      )}

      {isConnected && !playlistConfirmed && (
        <PlaylistResolutionPanel
          settingsStatus={connection.settingsStatus}
          settingsError={connection.settingsError}
          validation={connection.validation}
          validationLoading={connection.validationLoading}
          discovery={connection.discovery}
          discoveryLoading={connection.discoveryLoading}
          onFindPlaylistAgain={connection.findPlaylistAgain}
          onRetryValidation={connection.retryValidation}
          onRetryDiscovery={connection.retryDiscovery}
          onContinueDiscovery={connection.continueDiscovery}
        />
      )}

      {isConnected && playlistConfirmed && (
        <RecentVideosPanel
          videos={connection.videos}
          videosLoading={connection.videosLoading}
          selectedVideoId={connection.selectedVideoId}
          onSelectVideo={connection.selectVideo}
          onRetryLoadVideos={connection.retryLoadVideos}
        />
      )}

      {selectedVideo && (
        <div className="flex items-center gap-2 rounded-lg border border-primary-200 bg-primary-50/60 p-3 text-sm text-primary-800 dark:border-primary-300/20 dark:bg-primary-500/10 dark:text-primary-200">
          <PlayCircle size={16} />
          Vídeo selecionado para hoje: <strong>{selectedVideo.title}</strong>
        </div>
      )}
    </div>
  );
}

// Adaptador React fino sobre src/services/english/dailyVideoConnectionController.ts.
// Toda a lógica de orquestração (quando autorizar, validar, descobrir, carregar
// vídeos) já está no controller e é testada lá sem precisar renderizar nada —
// este hook só liga o controller ao ciclo de vida do componente (useEffect) e
// republica o estado via useState.

import { useEffect, useRef, useState } from 'react';
import {
  authorizeYoutube,
  authorizeYoutubeWithConsent,
  getYoutubeAuthState,
  preloadGoogleIdentityServices,
  subscribeToYoutubeAuthState,
} from '../services/youtubeAuth';
import {
  discoverPlaylistByTitle,
  getRecentPlaylistVideos,
  validatePersistedPlaylist,
} from '../services/youtubePlaylistDiscovery';
import {
  getYoutubePlaylistSettings,
  saveYoutubePlaylistSettings,
  touchYoutubePlaylistLastVerifiedAt,
} from '../services/english/youtubePlaylistSettingsRepository';
import {
  createDailyVideoConnectionController,
  type DailyVideoConnectionController,
  type DailyVideoConnectionState,
} from '../services/english/dailyVideoConnectionController';

const PRODUCTION_DEPS = {
  getYoutubeAuthState,
  subscribeToYoutubeAuthState,
  authorizeYoutube,
  authorizeYoutubeWithConsent,
  getYoutubePlaylistSettings,
  saveYoutubePlaylistSettings,
  touchYoutubePlaylistLastVerifiedAt,
  discoverPlaylistByTitle,
  validatePersistedPlaylist,
  getRecentPlaylistVideos,
};

export interface UseDailyVideoConnectionResult extends DailyVideoConnectionState {
  connect: () => Promise<void>;
  connectWithConsent: () => Promise<void>;
  retryDiscovery: () => Promise<void>;
  continueDiscovery: () => Promise<void>;
  findPlaylistAgain: () => Promise<void>;
  retryValidation: () => Promise<void>;
  retryLoadVideos: () => Promise<void>;
  selectVideo: (videoId: string) => void;
}

export function useDailyVideoConnection(userId: string | null): UseDailyVideoConnectionResult {
  const controllerRef = useRef<DailyVideoConnectionController | null>(null);
  if (!controllerRef.current) {
    // Criação do controller é pura (sem side effects) — a inscrição no estado
    // de autorização e o carregamento inicial só acontecem dentro do
    // useEffect abaixo, nunca durante a renderização.
    controllerRef.current = createDailyVideoConnectionController(PRODUCTION_DEPS);
  }
  const controller = controllerRef.current;
  const [state, setState] = useState<DailyVideoConnectionState>(() => controller.getState());

  useEffect(() => {
    // Carrega o script do GIS de forma passiva (sem popup, sem consentimento)
    // para deixar o botão de conectar pronto mais rápido — nunca abre a
    // janela de autorização sozinho.
    preloadGoogleIdentityServices().catch(() => undefined);
  }, []);

  useEffect(() => {
    const unsubscribe = controller.subscribe(setState);
    if (userId) void controller.initialize(userId);
    return () => {
      unsubscribe();
      controller.dispose();
    };
  }, [controller, userId]);

  return {
    ...state,
    connect: controller.connect,
    connectWithConsent: controller.connectWithConsent,
    retryDiscovery: controller.retryDiscovery,
    continueDiscovery: controller.continueDiscovery,
    findPlaylistAgain: controller.findPlaylistAgain,
    retryValidation: controller.retryValidation,
    retryLoadVideos: controller.retryLoadVideos,
    selectVideo: controller.selectVideo,
  };
}

// Orquestração da Etapa 3 (Inglês Diário — V1): autorização do YouTube,
// configuração/descoberta da playlist "SGP — Inglês" e vídeos recentes.
//
// Deliberadamente NÃO é um hook React — é um controlador simples (estado +
// ações) que recebe suas dependências por injeção (`DailyVideoConnectionDeps`).
// Isso existe porque o projeto não tem infraestrutura de teste de componentes
// React (não há @testing-library/react nas devDependencies, e nenhum teste
// existente no repositório renderiza um componente) — então toda a lógica que
// precisa ser testada (quando o popup pode abrir, quando a validação/descoberta
// roda, o que acontece em cada resultado) vive aqui, testável diretamente com
// funções e mocks, sem precisar renderizar nada. O hook React
// (src/hooks/useDailyVideoConnection.ts) é apenas um adaptador fino por cima
// deste controlador.
//
// Regra de segurança mais importante deste arquivo: NUNCA chama
// getInMemoryYoutubeAccessToken() nem aceita/repassa um accessToken — as
// funções injetadas (discoverPlaylistByTitle, validatePersistedPlaylist,
// getRecentPlaylistVideos) já resolvem o token internamente (Etapa 2). Este
// controlador só decide QUANDO chamá-las.

import type {
  DiscoveredPlaylist,
  PlaylistDiscoveryResult,
  PlaylistValidationResult,
  RecentVideosResult,
  YoutubeAuthState,
  YoutubePlaylistSettingsRow,
} from '../../types/dailyVideoEnglish';

export interface DailyVideoConnectionDeps {
  getYoutubeAuthState: () => YoutubeAuthState;
  subscribeToYoutubeAuthState: (listener: (state: YoutubeAuthState) => void) => () => void;
  authorizeYoutube: () => Promise<YoutubeAuthState>;
  authorizeYoutubeWithConsent: () => Promise<YoutubeAuthState>;
  getYoutubePlaylistSettings: (userId: string) => Promise<YoutubePlaylistSettingsRow | null>;
  saveYoutubePlaylistSettings: (
    userId: string,
    playlist: { playlistId: string; playlistTitle: string },
  ) => Promise<YoutubePlaylistSettingsRow>;
  touchYoutubePlaylistLastVerifiedAt: (userId: string) => Promise<YoutubePlaylistSettingsRow>;
  discoverPlaylistByTitle: (options?: {
    maxPages?: number;
    startPageToken?: string;
    previousMatches?: DiscoveredPlaylist[];
  }) => Promise<PlaylistDiscoveryResult>;
  validatePersistedPlaylist: (
    playlistId: string | null | undefined,
    options?: { maxPages?: number },
  ) => Promise<PlaylistValidationResult>;
  getRecentPlaylistVideos: (
    playlistId: string,
    options?: { maxPages?: number; limit?: number },
  ) => Promise<RecentVideosResult>;
}

export interface DailyVideoConnectionState {
  /** Estado público de autorização (src/services/youtubeAuth.ts) — nunca contém o token. */
  authState: YoutubeAuthState;
  settingsStatus: 'idle' | 'loading' | 'loaded' | 'error';
  /** Configuração da playlist já salva no Supabase — só metadados, nunca token. */
  settings: YoutubePlaylistSettingsRow | null;
  settingsError: string | null;
  validation: PlaylistValidationResult | null;
  validationLoading: boolean;
  discovery: PlaylistDiscoveryResult | null;
  discoveryLoading: boolean;
  videos: RecentVideosResult | null;
  videosLoading: boolean;
  selectedVideoId: string | null;
}

export interface DailyVideoConnectionController {
  getState: () => DailyVideoConnectionState;
  subscribe: (listener: (state: DailyVideoConnectionState) => void) => () => void;
  /** Deve ser chamada uma vez, ao montar a página — nunca abre popup por conta própria. */
  initialize: (userId: string) => Promise<void>;
  /** Só deve ser chamada a partir de um clique explícito do usuário. */
  connect: () => Promise<void>;
  /** Só deve ser chamada a partir de um NOVO clique explícito (ex.: "Solicitar permissão novamente"). */
  connectWithConsent: () => Promise<void>;
  retryDiscovery: () => Promise<void>;
  continueDiscovery: () => Promise<void>;
  findPlaylistAgain: () => Promise<void>;
  retryValidation: () => Promise<void>;
  retryLoadVideos: () => Promise<void>;
  selectVideo: (videoId: string) => void;
  dispose: () => void;
}

function initialState(deps: DailyVideoConnectionDeps): DailyVideoConnectionState {
  return {
    authState: deps.getYoutubeAuthState(),
    settingsStatus: 'idle',
    settings: null,
    settingsError: null,
    validation: null,
    validationLoading: false,
    discovery: null,
    discoveryLoading: false,
    videos: null,
    videosLoading: false,
    selectedVideoId: null,
  };
}

export function createDailyVideoConnectionController(
  deps: DailyVideoConnectionDeps,
): DailyVideoConnectionController {
  let userId: string | null = null;
  let state = initialState(deps);
  let unsubscribeAuth: (() => void) | null = null;
  const listeners = new Set<(state: DailyVideoConnectionState) => void>();

  function setState(patch: Partial<DailyVideoConnectionState>): void {
    state = { ...state, ...patch };
    for (const listener of listeners) listener(state);
  }

  async function loadVideosFor(playlistId: string): Promise<void> {
    setState({ videosLoading: true });
    const videos = await deps.getRecentPlaylistVideos(playlistId);
    setState({ videos, videosLoading: false });
    if (userId && videos.kind === 'videos') {
      // Melhor-esforço: não deve travar a tela de vídeos já carregada se falhar.
      deps.touchYoutubePlaylistLastVerifiedAt(userId).catch(() => undefined);
    }
  }

  async function runDiscovery(continuation?: {
    startPageToken: string;
    previousMatches: DiscoveredPlaylist[];
  }): Promise<void> {
    if (!userId) return;
    setState({ discoveryLoading: true, discovery: continuation ? state.discovery : null });
    const discovery = await deps.discoverPlaylistByTitle(
      continuation
        ? { startPageToken: continuation.startPageToken, previousMatches: continuation.previousMatches }
        : {},
    );
    setState({ discovery, discoveryLoading: false });

    if (discovery.kind === 'found') {
      // Só salva a configuração quando existir exatamente uma playlist
      // encontrada — nunca escolhe silenciosamente em caso de duplicidade.
      const saved = await deps.saveYoutubePlaylistSettings(userId, discovery.playlist);
      setState({ settings: saved });
      await loadVideosFor(saved.playlist_id);
    }
    // not_found / duplicate / incomplete / error: só exibidos, sem ação automática.
  }

  async function validateAndLoad(): Promise<void> {
    if (!userId) return;
    const playlistId = state.settings?.playlist_id;
    if (!playlistId) {
      await runDiscovery();
      return;
    }

    setState({ validationLoading: true, validation: null });
    const validation = await deps.validatePersistedPlaylist(playlistId);
    setState({ validation, validationLoading: false });

    if (validation.kind === 'valid') {
      await loadVideosFor(validation.playlist.playlistId);
    }
    // invalid_or_missing / inaccessible / auth_error / temporary_failure: a
    // configuração salva NÃO é apagada aqui — a interface oferece "Localizar
    // playlist novamente" (findPlaylistAgain) para o usuário decidir.
  }

  function onAuthStateChange(next: YoutubeAuthState): void {
    const wasAuthorized = state.authState.status === 'authorized';
    setState({ authState: next });
    if (!wasAuthorized && next.status === 'authorized') {
      void validateAndLoad();
    }
  }

  return {
    getState: () => state,
    subscribe(listener) {
      listeners.add(listener);
      return () => listeners.delete(listener);
    },

    async initialize(forUserId: string) {
      userId = forUserId;
      if (!unsubscribeAuth) {
        unsubscribeAuth = deps.subscribeToYoutubeAuthState(onAuthStateChange);
      }

      setState({ settingsStatus: 'loading', settingsError: null });
      try {
        const settings = await deps.getYoutubePlaylistSettings(forUserId);
        setState({ settings, settingsStatus: 'loaded' });
      } catch (err) {
        setState({
          settingsStatus: 'error',
          settingsError: err instanceof Error ? err.message : 'Erro ao carregar a configuração da playlist.',
        });
      }

      // "Playlist configurada" nunca é tratada como prova de autorização: só
      // validamos/descobrimos se JÁ existir uma sessão autorizada em memória
      // neste exato momento (ex.: o escopo já havia sido concedido antes,
      // nesta mesma execução da página).
      if (deps.getYoutubeAuthState().status === 'authorized') {
        await validateAndLoad();
      }
    },

    async connect() {
      await deps.authorizeYoutube();
    },

    async connectWithConsent() {
      await deps.authorizeYoutubeWithConsent();
    },

    async retryDiscovery() {
      await runDiscovery();
    },

    async continueDiscovery() {
      if (state.discovery?.kind !== 'incomplete' || !state.discovery.resumePageToken) return;
      await runDiscovery({
        startPageToken: state.discovery.resumePageToken,
        previousMatches: state.discovery.playlistsSeenSoFar,
      });
    },

    async findPlaylistAgain() {
      setState({ validation: null });
      await runDiscovery();
    },

    async retryValidation() {
      await validateAndLoad();
    },

    async retryLoadVideos() {
      const playlistId = state.validation?.kind === 'valid'
        ? state.validation.playlist.playlistId
        : state.settings?.playlist_id;
      if (playlistId) await loadVideosFor(playlistId);
    },

    selectVideo(videoId: string) {
      setState({ selectedVideoId: videoId });
    },

    dispose() {
      unsubscribeAuth?.();
      unsubscribeAuth = null;
      listeners.clear();
    },
  };
}

// Lógica pura (sem JSX) do rótulo do botão principal de ConnectYoutubeStep.tsx —

import type { YoutubeAuthStatus } from '../../../types/dailyVideoEnglish';

// Estados que representam uma reconexão de verdade — token que já existiu
// nesta execução e ficou inválido/expirado, ou uma tentativa de autorização
// que foi interrompida. Nenhum deles depende de haver ou não uma playlist
// salva no Supabase: a existência de youtube_playlist_settings NUNCA prova
// que existe (ou existiu) uma autorização ativa nesta página.
export const RECONNECT_STATUSES = new Set<YoutubeAuthStatus>([
  'token_invalid',
  'popup_closed',
  'popup_failed_to_open',
  'authorization_timeout',
]);

export function getPrimaryActionLabel(status: YoutubeAuthStatus): string {
  if (RECONNECT_STATUSES.has(status)) return 'Reconectar ao YouTube';
  if (status === 'authorization_denied' || status === 'unknown_error') return 'Tentar novamente';
  // 'ready' (nunca autorizado nesta execução) e qualquer outro caso residual.
  return 'Conectar ao YouTube';
}

            <p className="text-sm leading-6 text-surface-600 dark:text-surface-300">
              Você tem muitas playlists e ainda não terminamos de verificar todas. Isto ainda não é um resultado
              definitivo.
            </p>
            <Button type="button" variant="primary" icon={<RefreshCw size={16} />} onClick={onContinueDiscovery}>
              Continuar busca
            </Button>
          </CardBody>
        </Card>
      );
    }

    if (discovery.kind === 'error') {
      return (
        <Card>
          <CardHeader title="Não foi possível buscar sua playlist" icon={<AlertTriangle size={18} />} />
          <CardBody className="space-y-3">
            <p className="text-sm leading-6 text-surface-600 dark:text-surface-300">{describeApiError(discovery.error)}</p>
            <Button type="button" variant="secondary" icon={<RefreshCw size={16} />} onClick={onRetryDiscovery}>
              Tentar novamente
            </Button>
          </CardBody>
        </Card>
      );
    }

    // 'found' — transitório: os vídeos já devem estar carregando/carregados.
    return (
      <Card>
        <CardBody className="flex items-center gap-2 py-4 text-sm text-success-700 dark:text-success-300">
          <ListVideo size={16} />
          Playlist &ldquo;{discovery.playlist.playlistTitle}&rdquo; encontrada e configurada.
        </CardBody>
      </Card>
    );
  }

  return null;
}

  if (videos.kind === 'incomplete') {
    return (
      <Card>
        <CardHeader title="Vídeos recentes" icon={<Video size={18} />} />
        <CardBody className="space-y-3">
          <p className="text-sm leading-6 text-surface-600 dark:text-surface-300">
            Ainda estamos carregando os vídeos desta playlist (ela tem muitos itens) — este ainda não é um resultado
            definitivo.
          </p>
          <Button type="button" variant="secondary" icon={<RefreshCw size={16} />} onClick={onRetryLoadVideos}>
            Tentar novamente
          </Button>
        </CardBody>
      </Card>
    );
  }

  if (videos.kind === 'error') {
    return (
      <Card>
        <CardHeader title="Não foi possível carregar os vídeos" icon={<AlertTriangle size={18} />} />
        <CardBody className="space-y-3">
          <p className="text-sm leading-6 text-surface-600 dark:text-surface-300">{describeApiError(videos.error)}</p>
          <Button type="button" variant="secondary" icon={<RefreshCw size={16} />} onClick={onRetryLoadVideos}>
            Tentar novamente
          </Button>
        </CardBody>
      </Card>
    );
  }

  return (
    <Card>
      <CardHeader
        title="Vídeos recentes"
        subtitle="Selecione o vídeo que você estudou hoje. Ordenados pela data em que foram adicionados à playlist."
        icon={<Video size={18} />}
      />
      <CardBody>
        <div className="grid gap-3 sm:grid-cols-2 xl:grid-cols-3">
          {videos.videos.map(video => (
            <VideoCard
              key={video.videoId}
              video={video}
              selected={video.videoId === selectedVideoId}
              onSelect={onSelectVideo}
            />
          ))}
        </div>
      </CardBody>
    </Card>
  );
}
